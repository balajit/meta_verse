```rego
# Rego policies enforcing inheritance & state rules
package fsm.inheritance

import future.keywords.in
import future.keywords.every

default allow = false

# Allow resolution if no violations exist
allow {
    count(violations) == 0
}

# Rule 1: Child namespaces cannot mutate immutable primary key definitions
violations[msg] {
    some entity in input.ancestors
    child_pk := input.context.child_schema.entities[entity].primary_key
    parent_pk := input.context.parent_schema.entities[entity].primary_key
    child_pk != parent_pk
    msg := sprintf("Immutable Primary Key violation for entity '%s': child pk '%s' does not match parent pk '%s'", [entity, child_pk, parent_pk])
}

# Rule 2: Parent state transition invariants cannot be removed or altered in child FSMs
violations[msg] {
    some entity, fsm in input.context.parent_schema.fsms
    some state, transitions in fsm.immutable_states
    some transition in transitions

    # Check if child FSM breaks the transition invariant
    child_transitions := input.context.child_schema.fsms[entity].states[state]
    not transition in child_transitions
    msg := sprintf("FSM State invariant broken for entity '%s' in state '%s': missing required transition '%s'", [entity, state, transition])
}
```

```python
from typing import Any, Dict, Optional


class MetaBrainError(Exception):
    """Base exception for all Meta-Builder Brain runtime failures."""

    def __init__(self, message: str, context: Optional[Dict[str, Any]] = None) -> None:
        super().__init__(message)
        self.message = message
        self.context = context or {}


class ResolutionError(MetaBrainError):
    """Raised when namespace resolution fails along the hierarchy tree."""


class NamespaceCycleError(ResolutionError):
    """Raised when a cyclical parent-child inheritance reference is detected."""


class OpenBaoError(MetaBrainError):
    """Raised when secret retrieval or authentication with OpenBao fails."""


class ConfigurationError(MetaBrainError):
    """Raised when system startup configuration parameters are missing or invalid."""


class ValidationErrorTrace(MetaBrainError):
    """Captures Pydantic or Rego policy failures with diagnostic error traces."""

    def __init__(self, message: str, raw_trace: Dict[str, Any], context: Optional[Dict[str, Any]] = None) -> None:
        super().__init__(message, context)
        self.raw_trace = raw_trace
```

```python
                payload: Dict[str, Any] = response.json()

                    self.ANTHROPIC_API_KEY = secrets["ANTHROPIC_API_KEY"]
                if "QWEN_API_KEY" in secrets:
                    self.QWEN_API_KEY = secrets["QWEN_API_KEY"]
                if "DATABASE_URL" in secrets:
                    self.DATABASE_URL = secrets["DATABASE_URL"]

            except httpx.HTTPError as err:
                raise OpenBaoError(
                    f"Failed to fetch secret parameters from OpenBao at {url}: {str(err)}",
                    context={"status_code": getattr(err.response, "status_code", None)}
                ) from err
```

```python
from pydantic import BaseModel, Field, ConfigDict, field_validator


class AttributeType(str, Enum):

    INTEGER = "integer"
    FLOAT = "float"

    name: str = Field(..., description="Attribute name")
    data_type: AttributeType = Field(..., description="Target runtime data type")
    primary_key: bool = Field(default=False, description="Primary key indicator")
    nullable: bool = Field(default=True, description="Nullability constraint")

    to_state: str = Field(..., description="Target state identifier")
    trigger_event: str = Field(..., description="Event triggering state shift")

    model_config = ConfigDict(frozen=True)

    entity_name: str = Field(..., description="Bound entity identifier")
    initial_state: str = Field(..., description="Entrypoint state")
    states: List[str] = Field(..., description="Exhaustive list of valid FSM states")

        description="Map of protected states to required immutable child transitions"
    )
    transitions: List[TransitionModel] = Field(..., description="Allowed state transitions")


class EntityModel(BaseModel):
    model_config = ConfigDict(frozen=True)

    name: str = Field(..., description="Entity identifier")
    primary_key: str = Field(..., description="Designated primary key field name")
    attributes: Dict[str, AttributeModel] = Field(..., description="Map of field definitions")
    relationships: Dict[str, str] = Field(default_factory=dict, description="Map of relation fields to target entities")

    @field_validator("attributes")
    @classmethod
    def validate_pk_exists(cls, v: Dict[str, AttributeModel], info: Any) -> Dict[str, AttributeModel]:
        pk_field = info.data.get("primary_key")
        if pk_field and pk_field not in v:
            raise ValueError(f"Primary key field '{pk_field}' must be defined within entity attributes.")
        return v
```

```python
from typing import Any, Dict
import httpx

from meta_builder_brain.config.settings import settings
from meta_builder_brain.core.exceptions import ValidationErrorTrace


class OPAEvaluator:
    """Asynchronous HTTP client interfacing with localized Open Policy Agent sidecars."""

    def __init__(self, opa_url: str = settings.OPA_URL) -> None:
        self.opa_url = opa_url.rstrip("/")

    async def evaluate_policy(self, policy_path: str, input_payload: Dict[str, Any]) -> bool:
        """Submits synthesized manifests to OPA endpoint to execute Rego governance policy checks."""
        endpoint = f"{self.opa_url}/v1/data/{policy_path.lstrip('/')}"
        payload = {"input": input_payload}

        async with httpx.AsyncClient(timeout=5.0) as client:
            try:
                response = await client.post(endpoint, json=payload)
                response.raise_for_status()

                result: Dict[str, Any] = response.json().get("result", {})

                # Check top-level boolean decision or list of violations
                is_allowed: bool = result.get("allow", False)
                violations = result.get("violations", [])

                if not is_allowed or violations:
                    raise ValidationErrorTrace(
                        message=f"OPA policy validation failed for path '{policy_path}'. Violations count: {len(violations)}",
                        raw_trace={"violations": violations, "raw_result": result},
                        context={"policy_path": policy_path}
                    )

                return True

            except httpx.HTTPStatusError as err:
                raise ValidationErrorTrace(
                    message=f"OPA HTTP evaluation returned failure status code {err.response.status_code}",
                    raw_trace={"response_text": err.response.text},
                    context={"policy_path": policy_path, "status_code": err.response.status_code}
                ) from err
            except httpx.RequestError as err:
                raise ValidationErrorTrace(
                    message=f"Failed to connect to OPA sidecar at {endpoint}: {str(err)}",
                    raw_trace={"error": str(err)},
                    context={"policy_path": policy_path}
                ) from err
```

**`src/meta_builder_brain/synthesis/llm_client.py`**

```python
import asyncio
from typing import Any, Dict, Optional
import httpx
from anthropic import AsyncAnthropic

from meta_builder_brain.config.settings import settings
from meta_builder_brain.core.exceptions import MetaBrainError


class LLMClient:
    """Async client managing outbound requests to Anthropic and Qwen APIs with concurrency bounds."""

    def __init__(self, max_concurrent_requests: int = 5) -> None:
        self.semaphore = asyncio.Semaphore(max_concurrent_requests)
        self._anthropic_client: Optional[AsyncAnthropic] = None

    @property
    def anthropic(self) -> AsyncAnthropic:
        if not self._anthropic_client:
            if not settings.ANTHROPIC_API_KEY:
                raise MetaBrainError("Anthropic API key is not configured in settings.")
            self._anthropic_client = AsyncAnthropic(api_key=settings.ANTHROPIC_API_KEY)
        return self._anthropic_client

    async def generate_anthropic(
        self,
        prompt: str,
        system_prompt: str,
        model: str = "claude-3-5-sonnet-20241022",
        max_tokens: int = 4096,
        temperature: float = 0.0,
    ) -> str:
        async with self.semaphore:
            response = await self.anthropic.messages.create(
                model=model,
```

```python
                "text": page.get_text("text"),
                "blocks": page.get_text("blocks"),
            })
```

```python
import json
from typing import Any, Dict, Union
import yaml

from meta_builder_brain.core.types import ASTNode, ASTNodeProperty, PropertyGraph
from meta_builder_brain.core.exceptions import MetaBrainError


class SchemaParser:
    """Parses OpenAPI v3 and raw JSON Schema payloads into normalized property AST graphs."""

    def parse_to_ast(self, raw_schema: Union[str, Dict[str, Any]]) -> PropertyGraph:
        if isinstance(raw_schema, str):
            try:
                data = json.loads(raw_schema)
            except json.JSONDecodeError:
                data = yaml.safe_load(raw_schema)
        else:
            data = raw_schema

        nodes: Dict[str, ASTNode] = {}
        schemas = data.get("components", {}).get("schemas", {}) if "openapi" in data else data.get("$defs", data.get("definitions", {}))

        if not schemas and "properties" in data:
            schemas = {data.get("title", "RootEntity"): data}

        for entity_name, schema_body in schemas.items():
            properties: Dict[str, ASTNodeProperty] = {}
            required_fields = set(schema_body.get("required", []))

            for prop_name, prop_def in schema_body.get("properties", {}).items():
                properties[prop_name] = ASTNodeProperty(
                    type=prop_def.get("type", "string"),
                    required=(prop_name in required_fields),
                    is_primary_key=(prop_name in ["id", f"{entity_name.lower()}_id"]),
                    description=prop_def.get("description"),
                )

            nodes[entity_name] = ASTNode(
                name=entity_name,
                properties=properties,
            )

        return PropertyGraph(nodes=nodes, metadata={"source_version": data.get("openapi", "json-schema")})
```

```python
from typing import Any, Dict, Callable, Awaitable
from pydantic import ValidationError

from meta_builder_brain.core.exceptions import ValidationErrorTrace, MetaBrainError
from meta_builder_brain.validation.pydantic_models import NamespaceManifest


class SynthesisRetryLoop:
    """Executes automated feedback loops intercepting schema validation errors to re-prompt synthesis models."""

    def __init__(self, max_retries: int = 3) -> None:
        self.max_retries = max_retries

    async def execute_with_retry(
        self,
        synthesis_fn: Callable[[str], Awaitable[Dict[str, Any]]],
        initial_prompt: str,
    ) -> NamespaceManifest:
        current_prompt = initial_prompt

        for attempt in range(1, self.max_retries + 1):
            try:
                raw_result = await synthesis_fn(current_prompt)
                # Strict structural validation via Pydantic model
                validated_manifest = NamespaceManifest.model_validate(raw_result)
                return validated_manifest

            except (ValidationError, ValidationErrorTrace) as err:
                if attempt == self.max_retries:
                    raise ValidationErrorTrace(
                        message=f"Synthesis retry loop exhausted after {self.max_retries} attempts.",
                        raw_trace={"last_error": str(err)},
                        context={"attempts": attempt}
                    ) from err

                feedback_payload = self._construct_feedback_prompt(initial_prompt, str(err), attempt)
                current_prompt = feedback_payload

        raise MetaBrainError("Unexpected exit from retry loop.")

    def _construct_feedback_prompt(self, base_prompt: str, error_trace: str, attempt: int) -> str:
        return (
            f"{base_prompt}\n\n"
            f"--- REPAIR INSTRUCTION (Attempt {attempt}) ---\n"
            f"Your previous output failed validation with the following error:\n"
            f"{error_trace}\n\n"
            f"Correct the syntax and constraints to output ONLY a valid JSON structure."
        )
```

class HermesSynthesisAgent:
    """Single-agent orchestrator conducting two-pass entity extraction, state-machine binding,

    and cross-reference policy validation against OPA sidecars.
    """

    def __init__(
        self,
        llm_client: Optional[LLMClient] = None,

        namespace_id: str,
        tenant_id: str,
        tier: str,
        parent_manifest: Optional[NamespaceManifest] = None,

        Pass 1: Raw entity schema discovery and property extraction.
        Pass 2: FSM lifecycle synthesis, parent inheritance alignment, and OPA governance check.

        pass1_prompt = f"Extract structural entity model AST specs from standard description:\n{specification_text}"

        if parent_manifest:
            pass1_prompt += f"\n\nParent Ancestor Entities to preserve:\n{json.dumps(parent_manifest.entities, default=str)}"

            prompt=pass1_prompt,
            system_prompt=SCHEMA_EXTRACTION_SYSTEM_PROMPT,
            response_format=RAW_ENTITY_EXTRACTION_SCHEMA,

            raw_entities_data = json.loads(raw_entities_json)
        else:
            raw_entities_data = raw_entities_json

        # --- PASS 2: State Machine & Manifest Cross-Reference Validation ---
        async def pass2_synthesis_closure(prompt_feedback: str) -> Dict[str, Any]:
            pass2_prompt = (
                f"Synthesize the complete Namespace Manifest combining entities and state machine lifecycles.\n"
                f"Namespace ID: {namespace_id}\n"
                f"Parent ID: {parent_manifest.namespace_id if parent_manifest else 'None'}\n"

            await self.opa_evaluator.evaluate_policy(
                policy_path="fsm/inheritance",
                input_payload=opa_input,
            )

```python
from typing import List, Optional
from sqlalchemy import select, literal
from sqlalchemy.ext.asyncio import AsyncSession

from meta_builder_brain.compiler.store import AsyncSessionFactory, NamespaceManifestModel
from meta_builder_brain.core.exceptions import NamespaceCycleError, ResolutionError


class TreeResolver:
    """Resolves hierarchical namespace ancestor chains via PostgreSQL recursive CTE queries."""

    async def resolve_lineage(
        self, start_namespace_id: str, session: Optional[AsyncSession] = None
    ) -> List[NamespaceManifestModel]:
        """Executes recursive graph traversal from child namespace up to root Global tier."""
        session_owner = session is None
        async_session = session or AsyncSessionFactory()

        try:
            # Recursive CTE defining child-to-root resolution lineage
            base_cte = (
                select(
                    NamespaceManifestModel.id,
                    NamespaceManifestModel.parent_id,
                    NamespaceManifestModel.tenant_id,
                    NamespaceManifestModel.tier,
                    NamespaceManifestModel.revision,
                    NamespaceManifestModel.manifest_data,
                    NamespaceManifestModel.created_at,
                    NamespaceManifestModel.updated_at,
                    literal(1).label("depth"),
                )
                .where(NamespaceManifestModel.id == start_namespace_id)
                .cte(name="manifest_ancestors", recursive=True)
            )

            parent_alias = NamespaceManifestModel
            recursive_cte = select(
                parent_alias.id,
                parent_alias.parent_id,
                parent_alias.tenant_id,
                parent_alias.tier,
                parent_alias.revision,
                parent_alias.manifest_data,
                parent_alias.created_at,
                parent_alias.updated_at,
                (base_cte.c.depth + 1).label("depth"),
            ).join(base_cte, parent_alias.id == base_cte.c.parent_id)

            ancestor_chain = base_cte.union_all(recursive_cte)
            stmt = select(ancestor_chain).order_by(ancestor_chain.c.depth.asc())

            if session_owner:
                async with async_session as active_session:
                    result = await active_session.execute(stmt)
                    rows = result.all()
            else:
                result = await async_session.execute(stmt)
                rows = result.all()

            if not rows:
                raise ResolutionError(
                    f"Namespace target '{start_namespace_id}' could not be resolved in persistent hierarchy."
                )

            # Cycle detection and max depth safeguard
            visited_ids = set()
            lineage_models: List[NamespaceManifestModel] = []

            for row in rows:
                if row.id in visited_ids:
                    raise NamespaceCycleError(
                        f"Cyclical inheritance detected in namespace resolution tree at node '{row.id}'"
                    )
                visited_ids.add(row.id)

                lineage_models.append(
                    NamespaceManifestModel(
                        id=row.id,
                        parent_id=row.parent_id,
                        tenant_id=row.tenant_id,
                        tier=row.tier,
                        revision=row.revision,
                        manifest_data=row.manifest_data,
                        created_at=row.created_at,
                        updated_at=row.updated_at,
                    )
                )

            return lineage_models

        finally:
            if session_owner and async_session:
                await async_session.close()

    async def get_manifests_by_ids(
        self, namespace_ids: List[str], session: Optional[AsyncSession] = None
    ) -> List[NamespaceManifestModel]:
        """Bulk retrieves explicit namespace manifest models preserved in specified sequence."""
        session_owner = session is None
        async_session = session or AsyncSessionFactory()

        try:
            stmt = select(NamespaceManifestModel).where(NamespaceManifestModel.id.in_(namespace_ids))

            if session_owner:
                async with async_session as active_session:
                    result = await active_session.execute(stmt)
                    records = {m.id: m for m in result.scalars().all()}
            else:
                result = await async_session.execute(stmt)
                records = {m.id: m for m in result.scalars().all()}

            ordered_manifests = [records[nid] for nid in namespace_ids if nid in records]
            return ordered_manifests

        finally:
            if session_owner and async_session:
                await async_session.close()
```

```python
import asyncio
import time
from typing import Dict, Optional, Tuple

from meta_builder_brain.config.settings import settings
from meta_builder_brain.core.types import ResolutionContext, ResolutionPath


class CacheGuard:
    """Thread-safe, asyncio.Lock-guarded in-memory TTL cache for verified ResolutionPath models."""

    def __init__(self, ttl_seconds: int = settings.CACHE_TTL_SECONDS) -> None:
        self.ttl_seconds = ttl_seconds
        self._cache: Dict[str, Tuple[ResolutionPath, float]] = {}
        self._lock = asyncio.Lock()

    def _build_key(self, context: ResolutionContext) -> str:
        return f"{context.tenant_id}:{context.industry_domain}:{context.custom_namespace or ''}"

    async def get(self, context: ResolutionContext) -> Optional[ResolutionPath]:
        """Retrieves non-expired resolution paths from memory cache."""
        key = self._build_key(context)
        async with self._lock:
            if key in self._cache:
                path, timestamp = self._cache[key]
                if time.time() - timestamp < self.ttl_seconds:
                    return path.model_copy(update={"is_cached": True})
                # Evict expired entry
                del self._cache[key]
        return None

    async def set(self, context: ResolutionContext, path: ResolutionPath) -> None:
        """Stores verified resolution path with current timestamp mark."""
        key = self._build_key(context)
        async with self._lock:
            self._cache[key] = (path, time.time())

    async def invalidate(self, context: ResolutionContext) -> None:
        """Evicts a target context key from the cache store."""
        key = self._build_key(context)
        async with self._lock:
            self._cache.pop(key, None)

    async def clear(self) -> None:
        """Flushes all entries from memory cache."""
        async with self._lock:
            self._cache.clear()
```

```python
class FallbackChainResolver:

    def __init__(
        self,
        tree_resolver: Optional[TreeResolver] = None,
        cache_guard: Optional[CacheGuard] = None,

        """Resolves ancestor lineage, validates pairwise FSM/PK inheritance with OPA, and caches outcome."""
        # 1. Attempt cached path hit
        cached_path = await self.cache_guard.get(context)
        if cached_path:
            lineage_models = await self.tree_resolver.get_manifests_by_ids(cached_path.path)
            return cached_path, lineage_models

        # 2. Execute DB recursive CTE tree resolution
        lineage = await self.tree_resolver.resolve_lineage(target_namespace_id)
        if not lineage:
            raise ResolutionError(f"Namespace hierarchy resolution yielded empty set for '{target_namespace_id}'")

        # 3. Stepwise pairwise validation along ancestor chain (Child -> Parent)
        for i in range(len(lineage) - 1):
            child_manifest = lineage[i]
            parent_manifest = lineage[i + 1]

            opa_payload = {
                "context": {
                    "child_schema": child_manifest.manifest_data,
                    "parent_schema": parent_manifest.manifest_data,
                },
                "ancestors": list(child_manifest.manifest_data.get("entities", {}).keys()),
            }

            # Enforce immutable PK and FSM invariant policies via Rego
            await self.opa_evaluator.evaluate_policy(
                policy_path="fsm/inheritance", input_payload=opa_payload
            )

        # 4. Construct verified resolution path outcome
        path_ids = [m.id for m in lineage]
        effective_tier = NamespaceTier(lineage[0].tier)

        resolution_path = ResolutionPath(
            path=path_ids,
            tenant_id=context.tenant_id,
            effective_tier=effective_tier,
            is_cached=False,
        )

        # 5. Populate CacheGuard memory store
        await self.cache_guard.set(context, resolution_path)

        return resolution_path, lineage
```

```python
from typing import Any, Dict, List, Type
from sqlalchemy import Column, Float, Integer, String, Boolean, DateTime, JSON
from sqlalchemy.dialects.postgresql import UUID as PG_UUID
from sqlalchemy.orm import DeclarativeBase

from meta_builder_brain.compiler.store import Base, NamespaceManifestModel
from meta_builder_brain.validation.pydantic_models import AttributeType

# Mapping runtime attribute definitions to SQLAlchemy column types
TYPE_MAP = {
    AttributeType.STRING.value: String,
    AttributeType.INTEGER.value: Integer,
    AttributeType.FLOAT.value: Float,
    AttributeType.BOOLEAN.value: Boolean,
    AttributeType.DATETIME.value: DateTime(timezone=True),
    AttributeType.UUID.value: PG_UUID(as_uuid=True),
    AttributeType.JSON.value: JSON,
}


class DynamicMetaclassBuilder:
    """Constructs executable SQLAlchemy ORM model classes dynamically via type() metaclass factories."""

    def __init__(self, base_class: Type[DeclarativeBase] = Base) -> None:
        self.base_class = base_class
        self._compiled_registry: Dict[str, Type[DeclarativeBase]] = {}

    def compile_orm_models(
        self, lineage_manifests: List[NamespaceManifestModel]
    ) -> Dict[str, Type[DeclarativeBase]]:
        """Traverses lineage manifests from Root (Global) down to Custom overrides,

        merging entity schemas and dynamically instantiating ORM classes.
        """
        # Process in reverse sequence: Root (Global) -> Ancestors -> Child (Custom overlay)
        ordered_manifests = list(reversed(lineage_manifests))
        merged_entities: Dict[str, Dict[str, Any]] = {}

        for manifest_model in ordered_manifests:
            entities = manifest_model.manifest_data.get("entities", {})

            for entity_name, entity_def in entities.items():
                if entity_name not in merged_entities:
                    merged_entities[entity_name] = {
                        "primary_key": entity_def.get("primary_key"),
                        "attributes": {},
                    }

                # Overlay attributes from specific child extensions
                merged_entities[entity_name]["attributes"].update(
                    entity_def.get("attributes", {})
                )

        compiled_models: Dict[str, Type[DeclarativeBase]] = {}

        for entity_name, entity_spec in merged_entities.items():
            class_name = f"Dynamic_{entity_name}"
            table_name = f"dynamic_{entity_name.lower()}"

            # Check dynamic registry cache
            if class_name in self._compiled_registry:
                compiled_models[entity_name] = self._compiled_registry[class_name]
                continue

            class_dict: Dict[str, Any] = {
                "__tablename__": table_name,
                "__module__": "meta_builder_brain.compiler.dynamic",
            }

            pk_field = entity_spec.get("primary_key")
            attributes = entity_spec.get("attributes", {})

            for attr_name, attr_def in attributes.items():
                data_type_str = attr_def.get("data_type", "string")
                sa_type = TYPE_MAP.get(data_type_str, String)

                is_pk = (attr_name == pk_field) or attr_def.get("primary_key", False)
                nullable = attr_def.get("nullable", True) if not is_pk else False

                class_dict[attr_name] = Column(
                    sa_type,
                    primary_key=is_pk,
                    nullable=nullable,
                    default=attr_def.get("default_value"),
                )

            # Dynamic type() metaclass allocation
            orm_class = type(class_name, (self.base_class,), class_dict)
            self._compiled_registry[class_name] = orm_class
            compiled_models[entity_name] = orm_class

        return compiled_models
```

```python
@router.post("/ingest/raw", status_code=status.HTTP_201_CREATED)
async def ingest_raw_schema(
    payload: RawSchemaIngestRequest,
    db: AsyncSession = Depends(get_async_session)
) -> Dict[str, Any]:
    """Parses raw OpenAPI/JSON schemas, normalizes property trees, and saves namespace manifest."""
    try:
        schema_parser = SchemaParser()
        property_graph = schema_parser.parse_to_ast(payload.raw_payload)

        # Construct candidate manifest data
        manifest_data = {
            "namespace_id": payload.namespace_id,
            "parent_id": payload.parent_id,
            "tier": payload.tier.value,
            "version": 1,
            "entities": {
                name: {
                    "name": node.name,
                    "primary_key": next((k for k, v in node.properties.items() if v.is_primary_key), "id"),
                    "attributes": {
                        k: {"name": k, "data_type": v.type, "primary_key": v.is_primary_key, "nullable": not v.required}
                        for k, v in node.properties.items()
                    },
                    "relationships": {}
                }
                for name, node in property_graph.nodes.items()
            },
            "fsms": {}
        }

        manifest = NamespaceManifest.model_validate(manifest_data)

        # Persist to database
        db_model = NamespaceManifestModel(
            id=manifest.namespace_id,
            parent_id=manifest.parent_id,
            tenant_id=payload.tenant_id,
            tier=manifest.tier,
            revision=manifest.version,
            manifest_data=manifest.model_dump()
        )
        db.add(db_model)
        await db.commit()

        return {"status": "success", "namespace_id": manifest.namespace_id, "entities_count": len(manifest.entities)}

    except (ValidationErrorTrace, MetaBrainError) as err:
        raise HTTPException(status_code=status.HTTP_400_BAD_REQUEST, detail=str(err))

    except MetaBrainError as err:
```
